```python
# Analysis.py (Target: <500 lines)

from flask import Flask, render_template
from core.config_loader import load_config
from core.error_handler import register_error_handlers
from core.database import init_database
from core.dependencies import init_repositories, init_services
from api import upload_routes, chat_routes, cluster_routes, config_routes, history_routes

def create_app():
    """Application factory pattern"""
    app = Flask(__name__)

    # 1. Load configuration
    config = load_config()
    app.config.update(config)

    # 2. Initialize database
    db = init_database()

    # 3. Initialize dependencies
    repos = init_repositories(db)
    services = init_services(repos)

    # 4. Register blueprints
    app.register_blueprint(upload_routes.create_upload_blueprint(services['ticket_service']))
    app.register_blueprint(chat_routes.create_chat_blueprint(services['rag_service']))
    app.register_blueprint(cluster_routes.create_cluster_blueprint(services['cluster_service']))
    app.register_blueprint(config_routes.create_config_blueprint(repos['config_repo']))
    app.register_blueprint(history_routes.create_history_blueprint(repos['ticket_repo']))

    # 5. Register error handlers
    register_error_handlers(app)

    # 6. Static routes
    @app.route('/')
    @app.route('/index')
    def index():
        return render_template('index.html')

    return app

if __name__ == '__main__':
    app = create_app()
    app.run(debug=True, host='0.0.0.0', port=5000)
```
